class Solution:
    def corpFlightBookings(self, bookings, n):
        arr = [0] * n

        for first, last, seats in bookings:
            arr[first - 1] += seats
            if last < n:
                arr[last] -= seats

        print(arr)

        res = [0] * n
        res[0] = arr[0]

        for i in range(1, n):
            res[i] = res[i - 1] + arr[i]

        return res
