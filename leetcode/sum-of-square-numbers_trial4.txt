class Solution:
    def judgeSquareSum(self, c: int) -> bool:
        
        l, r = 0, int(c ** 0.5)

        while l <= r :
            sqrs =  (l * l) + (r * r)

            if sqrs == c:
                return True

            elif sqrs < c:
                l += 1

            else:
                r -= 1
        
        return False
