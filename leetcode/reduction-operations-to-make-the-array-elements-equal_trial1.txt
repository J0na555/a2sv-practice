class Solution:
    def reductionOperations(self, nums: List[int]) -> int:

        numbers = sorted(set(nums))
        num_map = Counter(nums)
        n = len(numbers)

        res = 0
        for i in range(n):
            
            res += num_map[numbers[i]] * i

        return res
