class Solution:
    def pivotIndex(self, nums: List[int]) -> int:
        # acc = 0
        # prefix_sum = []

        # for num in nums:
        #     acc += num
        #     prefix_sum.append(acc)
        
        # for i in range(1, len(prefix_sum)):
        #     if prefix_sum[i-1] == prefix_sum[-1] - prefix_sum[i]:
        #         return i
        #     else:
        #         return -1

        total_sum = sum(nums)
        left_sum = 0
        
        for i in range(len(nums)):
            if left_sum == total_sum - left_sum - nums[i]:
                return i
                
            left_sum += nums[i]
        
        return -1
