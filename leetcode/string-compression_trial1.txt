class Solution:
    def compress(self, chars: List[str]) -> int:

        # ["a","b","b","b","b","b","b","b","b","b","b","b","b"]

        # ab12 = 4

        # ["a","a","b","b","c","c","c"]

        # a2b2c3 = 6

        # get duplicate num
        # add the char to the result 
        # count the number of duplicates
        # add it to the result if its > 1
        



        l, res = 0, 0

        while l < len(chars):
            r = l

            while r < len(chars) and chars[r] == chars[l]:
                r += 1

            chars[res] = chars[l]
            res += 1

            count = r - l
            if count > 1:
                for num in str(count):
                    chars[res] = num
                    res += 1

            l = r

        return res
