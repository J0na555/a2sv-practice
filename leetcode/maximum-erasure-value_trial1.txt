class Solution:
    def maximumUniqueSubarray(self, nums: List[int]) -> int:
        left = 0
        right = 0
        nums_set = set()

        max_sum = float('-inf')
        win_sum = 0

        while right < len(nums):
            if nums[right] in nums_set:
                nums_set.remove(nums[left])
                win_sum -= nums[left]
                left += 1
            else:
                win_sum += nums[right]
                nums_set.add(nums[right])
                right += 1
                max_sum = max(max_sum, win_sum)
        return max_sum
