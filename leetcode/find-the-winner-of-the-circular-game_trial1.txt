class Solution:
    def findTheWinner(self, n: int, k: int) -> int:

        nums = []
        for i in range(1, n+1):
            nums.append(i)


        current_index = 0


        while len(nums) > 1:

            
            remove_index = (current_index + k-1) % len(nums)

            nums.pop(remove_index)

            current_index = remove_index
        
        return nums[0]
