class Solution:
    def characterReplacement(self, s: str, k: int) -> int:
        longest = 0
        count = {}
        l = 0
        res = 0
    
        
        for r in range(len(s)):
            count[s[r]] = 1 + count.get(s[r], 0)
            longest = max(longest, count[s[r]])
            
            window_size = r-l+1
            
            # if out of k, just increment 1 to left and decrement 1 from longest 
            if (r - l + 1) - longest > k:                
                count[s[l]] -= 1
                l += 1
            
            res = max(res, r-l+1)
        
        return res
