class Solution:
    def mySqrt(self, x: int) -> int:
        if x < 2:
            return x
        
        l, r = 0, x // 2
        res = 0

        while l < r:

            mid = (l + r + 1) // 2 
            sqr = mid * mid

            if sqr == x:
                return mid

            elif sqr < x:
                l = mid
                res = mid

            else:
                r = mid - 1
        
        return res
