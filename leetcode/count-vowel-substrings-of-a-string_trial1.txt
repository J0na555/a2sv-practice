class Solution:
    def countVowelSubstrings(self, word: str) -> int:

        vowels = {'a','e', 'i', 'o', 'u'}

        # iterate over the string and look for vowels 
        # if it is vowel add it to the seen arr
        # and check if the len of the arr is 5 
        # if its is increament count by 1
        # return count

        count = 0
        n = len(word)

        for i in range(n):
            seen = set()
            
            for j in range(i, n):

                if word[j] not in vowels:
                    break
                
                seen.add(word[j])

            
                if len(seen) == 5:
                    count += 1
        
        return count
