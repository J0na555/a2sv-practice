class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        left, right  = 0, 0
        window = []
        longest = 0

        while right < len(s):

            if s[right] in window:
                window = window[1:]
                left += 1
                continue

            window.append(s[right])

            longest = max(longest, len(window))
            right+=1

        return longest
