class Solution:
    def maxArea(self, height: List[int]) -> int:
        n = len(height)
        l, r = 0, n-1
        area = 0
        max_area = 0
        min_height = float('inf')

        while l < r:
            min_height = min(height[l], height[r])
            width = r-l

            area = min_height * width

            max_area = max(max_area, area)

            if height[l] < height[r]:
                l += 1
            else:
                r -= 1
        
        return max_area
